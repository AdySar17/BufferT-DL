// ============================================================
// BFT Demon List — Firestore Security Rules
// Versión: V24 (IDs Custom: gestión integrada en /levels/{id})
// ============================================================
//
// CAMBIOS RESPECTO A V23:
//
//  1. IDs Custom (campo `customIds` en /levels/{id})
//     - Las IDs Custom se almacenan como array en el documento del nivel.
//     - NO requieren colección separada: son un campo más de /levels/{id}.
//     - Los permisos existentes ya cubren este caso:
//         * Lectura pública:     allow read: if true  → Todos los usuarios ven las IDs.
//         * Solo Staff escribe:  allow update: if isStaff()  → Mod/Admin/Owner pueden
//           añadir, editar y eliminar IDs Custom (junto con el resto del documento).
//     - No se añaden nuevas reglas porque las existentes son suficientes y correctas.
//       Añadir reglas redundantes podría generar conflictos o comportamientos
//       inesperados.
//
// INSTRUCCIONES DE DESPLIEGUE:
//
// Ve a Firebase Console → Firestore Database → Reglas
// y reemplaza el contenido completo por el de este archivo.
// Luego pulsa "Publicar".
//
// ============================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers ────────────────────────────────────────────────
    function isAuth()       { return request.auth != null; }
    function meUid()        { return request.auth.uid; }
    function isOwnerOf(uid) { return isAuth() && meUid() == uid; }

    // Lee el rol del usuario autenticado desde /users/{uid}
    function myRole() {
      return isAuth()
        ? get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get("role","Default")
        : "Default";
    }

    function isMod()      { return myRole() == "Mod";   }
    function isAdmin()    { return myRole() == "Admin";  }
    function isOwner()    { return myRole() == "Owner";  }
    function isAdminish() { return isAdmin() || isOwner(); }
    function isStaff()    { return isMod() || isAdmin() || isOwner(); }

    // ── /users/{uid} ────────────────────────────────────────────
    match /users/{uid} {
      allow read: if true;

      allow create: if isOwnerOf(uid);

      allow update: if isOwnerOf(uid) &&
        !("role"   in request.resource.data.diff(resource.data).affectedKeys()) &&
        !("banned" in request.resource.data.diff(resource.data).affectedKeys());

      allow update: if isOwner();

      allow update: if isAdmin() &&
        (
          !("role" in request.resource.data.diff(resource.data).affectedKeys()) ||
          !(request.resource.data.role in ["Admin","Owner"])
        ) &&
        (
          !("banned" in request.resource.data.diff(resource.data).affectedKeys()) ||
          resource.data.get("role","Default") != "Owner"
        );

      allow update: if isMod() &&
        !("role"   in request.resource.data.diff(resource.data).affectedKeys()) &&
        !("banned" in request.resource.data.diff(resource.data).affectedKeys());

      allow delete: if isAdminish();
    }

    // ── /profiles/{uid} ─────────────────────────────────────────
    match /profiles/{uid} {
      allow read: if true;

      allow create: if isOwnerOf(uid) || isStaff();

      allow update: if isOwnerOf(uid) || isStaff();

      allow delete: if isAdminish();
    }

    // ── /levels/{id} ────────────────────────────────────────────
    // [V24] El campo `customIds` (array de IDs Custom) se gestiona
    //       aquí como un campo más del documento del nivel.
    //       - Lectura pública → todos ven las IDs Custom.
    //       - Solo Staff (Mod/Admin/Owner) puede crear, actualizar y
    //         eliminar niveles, incluyendo el campo `customIds`.
    //       - Usuarios normales: solo lectura (visualización).
    match /levels/{id} {
      allow read:   if true;
      allow create: if isStaff();
      // [V23] Staff puede update: necesario para re-shift de posiciones,
      //       campo `rewards` y, desde V24, campo `customIds`.
      allow update: if isStaff();
      allow delete: if isAdminish();
    }

    // ── /records/{id} ───────────────────────────────────────────
    match /records/{id} {
      allow read: if true;

      allow create: if isAuth() &&
        !get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get("banned", false);

      allow update: if isStaff() ||
        (isAuth() && resource.data.userId == meUid());

      allow delete: if isStaff() ||
        (isAuth() && resource.data.userId == meUid() && resource.data.status != "Accepted");
    }

    // ── /positionHistory/{id} ───────────────────────────────────
    match /positionHistory/{id} {
      allow read:   if true;
      allow create: if isStaff();
      allow update: if isAdminish();
      allow delete: if isAdminish();
    }

    // ── /changelog/{id} ─────────────────────────────────────────
    match /changelog/{id} {
      allow read:   if true;
      allow create: if isStaff();
      allow update: if isAdminish();
      allow delete: if isAdminish();
    }

    // ── /notifications/{id} ─────────────────────────────────────
    match /notifications/{id} {
      allow read:   if isAuth() && resource.data.userId == meUid();
      allow create: if isStaff();
      allow update: if isStaff() ||
        (isAuth() && resource.data.userId == meUid());
      allow delete: if isAdminish() ||
        (isAuth() && resource.data.userId == meUid());
    }

    // ── /banners/{id} ─────────────────────────────────────────────
    // [V20] Catálogo global de banners cosméticos.
    // [V23] Staff puede CREATE banners: necesario para el auto-banner
    //       que se genera al aceptar un reclamo de nuevo nivel.
    //       Update y Delete siguen restringidos a Admin/Owner.
    match /banners/{id} {
      allow read:   if true;
      allow create: if isStaff();
      allow update: if isAdminish();
      allow delete: if isAdminish();
    }

    // ── /badges/{id} ──────────────────────────────────────────────
    // [V20] Catálogo global de insignias (badges).
    match /badges/{id} {
      allow read:   if true;
      allow create: if isAdminish();
      allow update: if isAdminish();
      allow delete: if isAdminish();
    }

    // ── /user_cosmetics/{uid} ─────────────────────────────────────
    // [V21] Inventario cosmético de cada usuario.
    // [V22] Staff puede actualizar cosméticos de OTROS usuarios.
    match /user_cosmetics/{uid} {
      allow read: if true;

      allow create: if isOwnerOf(uid) &&
        request.resource.data.get("unlockedBanners", []).size() == 0 &&
        request.resource.data.get("unlockedBadges",  []).size() == 0;

      allow update: if isOwnerOf(uid) &&
        !("unlockedBanners" in request.resource.data.diff(resource.data).affectedKeys()) &&
        !("unlockedBadges"  in request.resource.data.diff(resource.data).affectedKeys());

      allow create: if isOwner();
      allow update: if isOwner();
      allow update: if isStaff() && request.auth.uid != uid;
      allow delete: if isOwner();
    }

  }
}
